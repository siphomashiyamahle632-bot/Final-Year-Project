  <script>
    function togglePassword() {
      const password = document.getElementById("password");
      const button = document.querySelector(".show-password");

      if (password.type === "password") {
        password.type = "text";
        button.textContent = "Hide";
      } else {
        password.type = "password";
        button.textContent = "Show";
      }
    }

    function showRecoveryMessage(event) {
      event.preventDefault();

      const message = document.getElementById("formMessage");
      message.className = "message error";
      message.textContent =
        "Password recovery will be connected when backend authentication is implemented.";
    }

    document.getElementById("loginForm").addEventListener("submit", function(event) {
      event.preventDefault();

      const message = document.getElementById("formMessage");

      message.className = "message";
      message.textContent = "";

      if (!this.checkValidity()) {
        message.className = "message error";
        message.textContent = "Please enter a valid email address and password.";
        this.reportValidity();
        return;
      }

      /*
        BACKEND INTEGRATION:
        During Flask integration this form will submit credentials
        to the server. The server will authenticate the user and
        redirect them according to their assigned role.
      */

      message.className = "message error";
      message.textContent =
        "Login details validated. Backend authentication and role-based dashboard redirection will be connected during integration.";
    });

    function toggleMenu() {
      const nav = document.querySelector(".nav-links");
      const menuButton = document.querySelector(".menu-button");
      const isOpen = nav.classList.contains("mobile-open");

      if (isOpen) {
        nav.classList.remove("mobile-open");
        nav.style.display = "none";
        menuButton.setAttribute("aria-expanded", "false");
      } else {
        nav.classList.add("mobile-open");
        nav.style.display = "flex";
        nav.style.flexDirection = "column";
        nav.style.position = "absolute";
        nav.style.top = "74px";
        nav.style.left = "0";
        nav.style.right = "0";
        nav.style.background = "#ffffff";
        nav.style.padding = "20px";
        nav.style.borderBottom = "1px solid #e4e7ec";
        nav.style.alignItems = "flex-start";
        nav.style.gap = "18px";
        menuButton.setAttribute("aria-expanded", "true");
      }
    }

    window.addEventListener("resize", function() {
      const nav = document.querySelector(".nav-links");
      const menuButton = document.querySelector(".menu-button");

      if (window.innerWidth > 900) {
        nav.classList.remove("mobile-open");
        nav.style.display = "";
        nav.style.flexDirection = "";
        nav.style.position = "";
        nav.style.top = "";
        nav.style.left = "";
        nav.style.right = "";
        nav.style.background = "";
        nav.style.padding = "";
        nav.style.borderBottom = "";
        nav.style.alignItems = "";
        nav.style.gap = "";
        menuButton.setAttribute("aria-expanded", "false");
      }
    });
  </script>
